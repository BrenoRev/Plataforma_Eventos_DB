SQL> -- Resumo por evento: criador, ingressos emitidos e vendidos, receita.

  2  SELECT e.id_evento,
  3         e.nome                                         AS evento,
  4         e.dt_realizacao,
  5         e.status,
  6         c.tipo_criador,

  8         COUNT(i.num_ingresso)                          AS emitidos,

 13    LEFT JOIN Empresa em ON em.id_criador = c.id_criador
 14    LEFT JOIN Produtor pr ON pr.id_criador = c.id_criador
 15    LEFT JOIN Usuario u  ON u.cpf         = pr.cpf
 16    LEFT JOIN Ingresso i ON i.id_evento   = e.id_evento
 17   GROUP BY e.id_evento, e.nome, e.dt_realizacao, e.status, c.tipo_criador,
