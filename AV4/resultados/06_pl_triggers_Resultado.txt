SQL> -- 4. Trigger de linha: regras da venda de um ingresso.
SQL> -- Dispara quando um ingresso livre recebe pedido, venha a venda da package ou
SQL> -- de um UPDATE direto. Bloqueia evento cancelado ou já realizado e pedido feito
SQL> -- antes da abertura da oferta. Depois grava em preco_pago o preço atual da
SQL> -- oferta (minimundo: o preço pago é o da oferta no dia da compra).
SQL> -- Lê Evento, Oferta e Pedido, nunca Ingresso, então não cai em tabela mutante.
SQL> CREATE OR REPLACE TRIGGER trg_ingresso_venda
  2  BEFORE UPDATE OF id_pedido ON Ingresso
  3  FOR EACH ROW
  4  WHEN (OLD.id_pedido IS NULL AND NEW.id_pedido IS NOT NULL)
  5  DECLARE
  6      v_status     Evento.status%TYPE;
  7      v_dt_evento  Evento.dt_realizacao%TYPE;
  8      v_preco      Oferta.preco%TYPE;
  9      v_abertura   Oferta.dt_abertura_venda%TYPE;
 10      v_dt_pedido  Pedido.dt_realizacao%TYPE;
 11  BEGIN
 12      SELECT e.status, e.dt_realizacao, o.preco, o.dt_abertura_venda
 13        INTO v_status, v_dt_evento, v_preco, v_abertura
 14        FROM Oferta o
 15        JOIN Evento e ON e.id_evento = o.id_evento
 16       WHERE o.id_evento = :NEW.id_evento
 17         AND o.id_setor  = :NEW.id_setor
 18         AND o.id_lote   = :NEW.id_lote;
 19  
 20      SELECT dt_realizacao INTO v_dt_pedido FROM Pedido WHERE id_pedido = :NEW.id_pedido;
 21  
 22      IF v_status = 'CANCELADO' THEN
 23          RAISE_APPLICATION_ERROR(-20030, 'Evento cancelado: venda bloqueada');
 24      ELSIF v_status = 'ENCERRADO' OR TRUNC(v_dt_pedido) > v_dt_evento THEN
 25          RAISE_APPLICATION_ERROR(-20031, 'Evento ja realizado: venda encerrada');
 26      ELSIF TRUNC(v_dt_pedido) < v_abertura THEN
 27          RAISE_APPLICATION_ERROR(-20032, 'Venda desta oferta abre em ' || TO_CHAR(v_abertura, 'DD/MM/YYYY'));
 28      END IF;
 29  
 30      :NEW.preco_pago := v_preco;
 31  END trg_ingresso_venda;
 32  /
